\begin{frame}\frametitle{Classificação de Sinais}

\begin{itemize}
    \item O que é um \textbf{Sinal}? 
\end{itemize}
\vspace{-0.5cm}
        \begin{columns}
            \begin{column}{0.3\textwidth}
                \begin{block}{Comunicações \cite{oppenheim1997signals}}
                    {\small Entende-se que um sinal é uma sequência de estados em um sistema de comunicação que codifica uma mensagem. A definição pode mudar de acordo com o contexto em que se está trabalhando.}
                \end{block}
            \end{column}
            \begin{column}{0.33\textwidth}
                \begin{block}{Digital \cite{cis307signals}}
                    {\small Um sinal é uma forma limitada de comunicação entre processos utilizada em sistemas Unix e outros sistemas operacionais.}
                \end{block}
                \begin{block}{Informação \cite{proakis2001digital}}
                {\small Um sinal é a representação de informações em uma forma física ou matemática para processamento}
                    
                \end{block}
            \end{column}
            \begin{column}{0.35\textwidth}
                \begin{block}{Física \cite{smith1997scientist}}
                    {\small Sinais são manifestações físicas de eventos ou processos, capturadas por sensores e processadas para análise ou decisão.}
                \end{block}
                \begin{block}{Transmissão \cite{shannon1948mathematical}}
                    {\small Sinal é a sequência de símbolos ou grandezas portadoras de informação em um canal de comunicação}
                \end{block}
            \end{column}
        \end{columns}
\end{frame}
