\begin{frame}{Indústria}
    \begin{itemize}
        \item Empresas industriais usam o gêmeo digital para \textbf{analisar, monitorar e gerenciar} seus sistemas de forma virtual
        \item Apoia a \textbf{previsão} de comportamentos futuros e de anomalias antes que ocorram
        \item Pilar da Indústria 4.0: integra TI, engenharia mecânica, eletrônica, mecatrônica e automação em um só modelo
        \item Depende de dados homogêneos (\textit{homogenization}) e de um vínculo contínuo e bidirecional entre o objeto físico e sua réplica digital
    \end{itemize}
    \begin{exampleblock}{Benefício central}
        Decisões industriais passam a ser guiadas por dados em tempo real, reduzindo custos operacionais e o tempo de resposta a falhas.
    \end{exampleblock}
\end{frame}

\begin{frame}{Óleo e Gás}
    \begin{itemize}
        \item Setor com ambientes de trabalho hostis e perigosos: o gêmeo digital ajuda a antecipar falhas e reduzir riscos de catástrofes
        \item Aplicações ao longo da cadeia:
        \begin{itemize}
            \item Planejamento e otimização do processo de perfuração
            \item Monitoramento de desempenho de campos petrolíferos
            \item Simulação e análise de dados de produção
            \item Segurança do pessoal de campo (\textit{wearables}, HUD, tags BLE)
            \item Manutenção preditiva de equipamentos
        \end{itemize}
        \item Vantagens: eficiência de produção, conformidade regulatória, redução de custos e maior segurança operacional
    \end{itemize}
\end{frame}

\begin{frame}{Energia}
    \begin{itemize}
        \item Gêmeos digitais aplicados a turbinas, geradores e demais equipamentos de geração de energia, otimizando manutenção e ciclo de vida
        \item Ajudam a acelerar a transição para fontes mais limpas e sistemas elétricos mais flexíveis e seguros
    \end{itemize}
    \begin{exampleblock}{Exemplos reais}
        \begin{itemize}
            \item \textbf{GE Digital Wind Farm}: gêmeo digital de parques eólicos que ajusta o projeto de cada turbina às condições específicas do local de instalação
            \item \textbf{Hitachi Energy -- IdentiQ}: gêmeo digital para sistemas de corrente contínua de alta tensão (HVDC) e qualidade de energia, acelerando a transição para um futuro neutro em carbono
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{Saúde -- Gestão Hospitalar}
    \begin{itemize}
        \item \textbf{Gestão de fluxo hospitalar}: réplica virtual do hospital permite testar mudanças de estratégia operacional, escala de equipe e modelos de atendimento antes de aplicá-los na prática
        \item \textbf{Fabricação de produtos médicos} (ex.: vacinas): monitoramento remoto do processo produtivo, com redução de desperdício
    \end{itemize}
    \begin{exampleblock}{Exemplos reais}
        \begin{itemize}
            \item \textbf{GE -- Hospital of Future (HoF)}: suíte de simulação para prever a operação hospitalar e o cuidado ao paciente
            \item \textbf{Mater Hospital (Dublin)} com Siemens Healthineers: gêmeo digital do departamento de radiologia $\rightarrow$ menor tempo de espera, maior utilização de equipamentos e menor custo de pessoal
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{Saúde -- Aplicações Clínicas}
    \begin{itemize}
        \item \textbf{Gêmeo digital cardiovascular}: análise inversa com redes neurais recorrentes (LSTM) detecta aneurisma da aorta abdominal com acurácia de 99,91\%
        \item \textbf{Pré-planejamento cirúrgico}: simulação prévia do procedimento em modelo do paciente, treinamento de equipes e redução do risco cirúrgico
        \item \textbf{Tratamento personalizado}: modelos \textit{Grey/Black Box} aplicados a câncer de útero e esclerose múltipla, refinando diagnóstico e terapia
        \item \textbf{COVID-19}: gêmeos digitais pessoais integram prontuário eletrônico, \textit{wearables} e dados de IoT para detecção precoce de sintomas
    \end{itemize}
\end{frame}

\begin{frame}{Cidades Inteligentes}
    \begin{itemize}
        \item \textbf{Gestão de tráfego}: a \textbf{Aimsun} simula o deslocamento de pessoas e veículos para identificar gargalos e testar soluções de fluxo
        \item \textbf{Construção}: gêmeos digitais orientam projeto, obra e operação de edificações urbanas
        \item \textbf{Monitoramento estrutural (SHM)}: pontes e prédios acompanhados em tempo real para prevenir falhas
        \item \textbf{Saúde}: réplicas de hospitais e do sistema de atendimento da cidade
        \item \textbf{Drenagem}: o motor \textbf{PipeDream} combina um solver hidráulico (equações de Saint-Venant) com filtro de Kalman para prever enchentes em tempo real
        \item \textbf{Rede elétrica}: gêmeos digitais de qualidade de energia, como o IdentiQ da Hitachi Energy
    \end{itemize}
\end{frame}

\begin{frame}{Manufatura}
    \begin{itemize}
        \item Gêmeos digitais de máquinas-ferramenta oferecem dados em tempo real ao longo de toda a produção (inclusive para clientes, em contexto B2B)
        \item Modelo \textbf{GHOST} (Geometria -- Histórico -- Atributos -- Instantâneos -- Topologia): arquitetura para sistemas flexíveis de manufatura (FMS)
        \item Abordagem da \textbf{Siemens}: modelo unificado com gêmeo digital de \textit{produto}, de \textit{produção} e de \textit{desempenho}, conectando fábrica física e virtual em tempo real
        \item Otimização do tempo de usinagem e da qualidade superficial em máquinas-ferramenta
        \item Diagnóstico e predição de falhas em produtos eletromecânicos
    \end{itemize}
\end{frame}

\begin{frame}{Transporte}
    \begin{exampleblock}{Aviação}
        \begin{itemize}
            \item \textbf{Rolls-Royce}: gêmeos digitais de motores aeronáuticos para manutenção preventiva
            \item \textbf{GE Aviation} (Azure Digital Twin): estado da aeronave e de seus componentes acompanhado em tempo real, antecipando reparos
        \end{itemize}
    \end{exampleblock}
    \begin{exampleblock}{Setor automotivo}
        \begin{itemize}
            \item \textbf{Bridgestone} -- \textit{Virtual Tire Modelling}: reduz o número de protótipos físicos de pneus
            \item \textbf{Tesla}: cada veículo vendido possui um gêmeo digital, usado para orientar atualizações de software via dados dos sensores
        \end{itemize}
    \end{exampleblock}
    \begin{itemize}
        \item \textbf{Cadeia de suprimentos}: identificação de gargalos logísticos com gêmeos digitais
    \end{itemize}
\end{frame}

\begin{frame}{Construção Civil}
    \begin{itemize}
        \item Atua nas três fases do ciclo de vida da obra:
        \begin{itemize}
            \item \textbf{Projeto}: testa o impacto de mudanças de design antes da execução
            \item \textbf{Construção}: monitoramento remoto do canteiro, antecipando erros e riscos
            \item \textbf{Operação e manutenção}: acompanha a saúde estrutural do edifício já construído
        \end{itemize}
        \item Estende o \textbf{BIM} (Building Information Modeling) ao acrescentar fluxo de dados bidirecional entre o ativo físico e o modelo digital
        \item \textbf{Segurança do trabalhador}: identifica riscos em tempo real no canteiro de obras
    \end{itemize}
    \begin{exampleblock}{Exemplos reais}
        \textbf{Lendlease Group} testou a viabilidade de torres de 28--29 andares em madeira sustentável; a \textbf{CadMakers} projetou um edifício híbrido de madeira maciça de 18 andares -- ambos usando gêmeos digitais na fase de projeto.
    \end{exampleblock}
\end{frame}

\begin{frame}{Manutenção Preditiva}
    \begin{itemize}
        \item Combina IoT e gêmeo digital para substituir componentes \textit{just-in-time}: nem cedo demais (desperdício), nem tarde demais (falha)
        \item Equilibra manutenção corretiva e preventiva, estendendo a vida útil dos ativos
        \item Tema transversal a todos os setores desta seção:
        \begin{itemize}
            \item Óleo e gás: perfuração e campos de produção
            \item Aviação: motores e estruturas da aeronave
            \item Construção/cidades inteligentes: monitoramento estrutural (SHM) de pontes e edifícios
        \end{itemize}
        \item Resultado: menos paradas não planejadas, menor custo de mão de obra e maior segurança
    \end{itemize}
\end{frame}
