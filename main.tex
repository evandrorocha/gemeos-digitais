% --- Slide 3.1: Visão Geral das Classificações ---
\begin{frame}{Tipos de Gêmeos Digitais -- Visão Geral}
    Os capítulos 1--3 do livro de Vohra \cite{vohra2022dt} apresentam \textbf{três formas complementares} de classificar gêmeos digitais:

    \vspace{0.6em}
    \begin{columns}[T]
        \begin{column}{0.32\textwidth}
            \begin{block}{Taxonomia de Grieves}
                Classifica pela \textbf{relação} com o ativo físico:\\[0.3em]
                DTP, DTI, DTA e DTE
            \end{block}
        \end{column}
        \begin{column}{0.32\textwidth}
            \begin{block}{Por Granularidade}
                Classifica pelo \textbf{escopo} do que é modelado:\\[0.3em]
                Componente $\to$ Produto $\to$ Sistema $\to$ Processo
            \end{block}
        \end{column}
        \begin{column}{0.32\textwidth}
            \begin{block}{Nível de Integração}
                Classifica pelo \textbf{fluxo de dados} entre físico e virtual:\\[0.3em]
                Modelo $\to$ Sombra $\to$ Gêmeo
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \footnotesize
    Fonte: Grieves \& Vickers \cite{grieves2017dt}; Vohra \cite{vohra2022dt}, caps.\ 1--3; Fuller et al.\ \cite{fuller2020dt}.
\end{frame}

% --- Slide 3.2: Taxonomia de Grieves (DTP / DTI) ---
\begin{frame}{Taxonomia de Grieves -- DTP e DTI}
    \footnotesize
    Proposta por Michael Grieves (2002) e formalizada com John Vickers (NASA, 2010) \cite{grieves2017dt}:

    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{DTP -- Digital Twin Prototype}
                \footnotesize
                \begin{itemize}\setlength\itemsep{1pt}
                    \item Existe \textbf{antes} do ativo físico
                    \item Informações para \textbf{descrever e produzir} o ativo real
                    \item Modelo 3D anotado, especificações de materiais, \emph{bill of processes}, \emph{bill of disposal}
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{DTI -- Digital Twin Instance}
                \footnotesize
                \begin{itemize}\setlength\itemsep{1pt}
                    \item Vinculado a um ativo físico \textbf{específico} por toda sua vida útil
                    \item Modelo 3D com GD\&T, histórico de fabricação, testes, registros de manutenção
                    \item Dados de sensores: passados, presentes e preditos
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}

    \vspace{0.3em}
    \centering
    \scriptsize \textit{``DTPs devem existir para todos os produtos sofisticados; DTIs existem apenas quando é importante ter informação ao longo de toda a vida do produto.''} -- Grieves \cite{grieves2017dt}
\end{frame}

% --- Slide 3.3: Taxonomia de Grieves (DTA / DTE) ---
\begin{frame}{Taxonomia de Grieves -- DTA e DTE}
    \begin{columns}[T]
        \begin{column}{0.45\textwidth}
            \begin{block}{DTA -- Digital Twin Aggregate}
                \footnotesize
                \begin{itemize}\setlength\itemsep{1pt}
                    \item Agregação de \textbf{todos} os DTIs
                    \item Análise macro e correlação entre múltiplas instâncias
                    \item Ex: frota de aeronaves, parque eólico
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.50\textwidth}
            \begin{block}{DTE -- Digital Twin Environment}
                \footnotesize
                Espaço integrado multi-domínio para operar os gêmeos digitais:
                \begin{description}\setlength\itemsep{1pt}
                    \item[\textcolor{dexpiBlue}{Preditivo}] Prever comportamento e desempenho futuro
                    \item[\textcolor{dexpiBlue}{Interrogativo}] Consultar históricos passados e presentes
                \end{description}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.3em}
    \centering
    \begin{tikzpicture}[

                     minimum width=1.6cm, minimum height=0.6cm, font=\scriptsize\bfseries,

    ]

        \node[box] (dtp) at (-3,0) {DTP};

        \node[box] (dti2) at (0.8,0) {DTI$_2$};
        \node[box, minimum width=0.8cm] (dots) at (2.2,0) {...};
        \node[box] (dtin) at (3.4,0) {DTI$_n$};

    \end{tikzpicture}
\end{frame}

% --- Slide 3.6: Tabela Comparativa dos Níveis ---
\begin{frame}{Modelo $\times$ Sombra $\times$ Gêmeo -- Comparação}
    \vspace{0.3em}
    \centering
    \footnotesize
    \begin{tabular}{l c c c}
        \toprule
        \textbf{Característica} & \textbf{Modelo Digital} & \textbf{Sombra Digital} & \textbf{Gêmeo Digital} \\
        \midrule
        Fluxo de dados      & Nenhum           & Unidirecional          & \textbf{Bidirecional} \\
        Atualização automática & Não             & Sim (físico$\to$digital) & Sim (ambas direções)  \\
        Reflete estado real  & Não              & Sim                    & Sim \\
        Atua sobre o físico  & Não              & Não                    & \textbf{Sim} \\
        Exemplo              & Modelo CAD 3D    & Dashboard IoT          & Controle autônomo \\
        \bottomrule
    \end{tabular}

    \vspace{0.8em}
    \begin{alertblock}{Atenção}
        Muitos sistemas apresentados como ``gêmeos digitais'' na indústria são, na verdade, \textbf{sombras digitais} -- recebem dados do ativo físico, mas \textbf{não enviam comandos} de volta \cite{fuller2020dt}.
    \end{alertblock}
\end{frame}

% --- Slide 3.7: Ferramentas do Conhecimento (Cap. 2) ---
\begin{frame}{Gêmeo Digital como Ferramenta de Conhecimento}
    Grieves \cite{grieves2014whitepaper} identifica que o gêmeo digital suporta três ferramentas fundamentais do conhecimento humano:

    \vspace{0.5em}
    \begin{columns}[T]
        \begin{column}{0.31\textwidth}
            \begin{block}{Conceituação}
                Visualizar o comportamento do ativo de forma \textbf{intuitiva e visual}, ao invés de ler relatórios tabulares.
            \end{block}
        \end{column}
        \begin{column}{0.31\textwidth}
            \begin{block}{Comparação}
                Comparar o estado \textbf{ideal} com o \textbf{real}, utilizando \emph{tolerance corridors} para identificar desvios.
            \end{block}
        \end{column}
        \begin{column}{0.31\textwidth}
            \begin{block}{Colaboração}
                Compartilhar a mesma visualização com \textbf{múltiplos especialistas} em locais remotos, simultaneamente.
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \footnotesize
    Fonte: Grieves \cite{grieves2014whitepaper}; apresentado no Cap.\ 2 de Vohra \cite{vohra2022dt}.
\end{frame}

% --- Slide 3.8: Resumo da Seção ---
\begin{frame}{Tipos de Gêmeos Digitais -- Resumo}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Taxonomia de Grieves}
                \footnotesize
                \begin{description}\setlength\itemsep{0pt}
                    \item[DTP] Protótipo (pré-fabricação)
                    \item[DTI] Instância (ativo específico)
                    \item[DTA] Agregado (todos os DTIs)
                    \item[DTE] Ambiente de operação
                \end{description}
            \end{block}

            \vspace{0.2em}
            \begin{exampleblock}{Nível de Integração}
                \footnotesize
                \begin{description}\setlength\itemsep{0pt}
                    \item[Modelo] Sem fluxo de dados
                    \item[Sombra] Fluxo unidirecional
                    \item[Gêmeo] Fluxo bidirecional
                \end{description}
            \end{exampleblock}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Por Granularidade}
                \footnotesize
                \begin{description}\setlength\itemsep{0pt}
                    \item[Parts] Componente individual
                    \item[Product] Conjunto de componentes
                    \item[System] Conjunto de produtos
                    \item[Process] Manufatura completa
                \end{description}
            \end{block}

            \vspace{0.2em}
            \begin{alertblock}{Ponto-chave}
                \footnotesize
                As três classificações são \textbf{ortogonais}: um mesmo gêmeo digital pode ser um DTI, do tipo Process Twin, com integração bidirecional.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}
