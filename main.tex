\section{Conceito de Gêmeos Digitais}
%------------------------------------------------

\begin{frame}[plain]

    {\LARGE\textbf{Conceito de Gêmeos Digitais}}\\[0.6em]
    {\footnotesize Baseado nos capítulos 1--3 de Vohra (2023)}
    \vfill
\end{frame}

\begin{frame}{1. Conceito de Gêmeos Digitais}
Um Gêmeo Digital (\textit{Digital Twin}) é o modelo virtual de um 'ativo' físico, feito para representar de maneira fiel e em forma digital o estado atual do ativo para agregação de dados, monitoramento e experimentação.
\begin{itemize}
    \item Ativo: Uma peça/parte, um produto, um sistema ou um processo (industrial).
    \item Assemelha-se, age e comporta-se como o ativo real.
    \item O Gêmeo Digital deve estar conectado ao ativo físico para \textbf{RASTREAR} e \textbf{IMITAR} o ativo \textbf{EM TEMPO REAL}.
\end{itemize}
    
\end{frame}

\begin{frame}{Outras Definições}
\begin{itemize}
    \item Modelo inteligente e \textbf{evolutivo}, a imagem digital de uma entidade ou processo físico. Coleções de artefatos conectados representando virtualmente um sistema \textbf{ao longo de seu ciclo de vida}.
    \item Glaessgen Stargel: “Um gêmeo digital é uma simulação abrangente — multifísica, multiescala e probabilística — de um produto complexo que combina os melhores modelos físicos disponíveis, atualizações de sensores, etc., para espelhar o ciclo de vida de seu gêmeo correspondente.”
\end{itemize}
\end{frame}

\begin{frame}{Tipos de Representantes Digitais}
\begin{itemize}
    \item Modelo Digital: Representação digital e estática do ativo
    \item \textit{Sombra} Digital: Modelo digital que se atualiza automaticamente pelos \textbf{dados recebidos do ativo}.
    \item Gêmeo Digital: Modelo com comunicação \textbf{bidirecional} com o ativo.
\end{itemize}
\begin{figure}
    \centering
    \includegraphics[width=0.7\linewidth]{figures/digital_reps.png}
    \caption{https://www.flexcon.it/news-events/digital-twin-vs-digital-shadow-vs-digital-model/}
\end{figure}

\end{frame}

\begin{frame}{Qualidades do Gêmeo Digital}
O Gêmeo Digital se diferencia dos demais modelos a partir de suas principais características:
\begin{itemize}
    \item Rastreamento do ativo em tempo real.
    \item Replicação de alta fidelidade dos aspectos do ativo.
    \item Histórico digital do ativo.
    \item Habilidade de predição do comportamento (elétrico, mecânico, químico...) do ativo. 
\end{itemize}
    
\end{frame}

\begin{frame}{2. Arquitetura de um Gêmeo Digital}
O núcleo do Gêmeo Digital é contruído a partir das principais características do ativo que se deseja rastrear.

Considerando que é possível ter múltiplas instâncias de um objeto:
\begin{itemize}
    \item Propriedades únicas como ID.
    \item Processo do Modelo (Ex: Máquina de Estados).
    \item Detalhes físicos, modelo 3D, BOM.
    \item Componentes e peças substituíveis.
    \item Telemetria/Sensores.
    \item Dados Calculados.
    \item Contexto.
\end{itemize}


\end{frame}

% --- Slide 3.1: Visão Geral das Classificações ---
\begin{frame}{Tipos de Gêmeos Digitais -- Visão Geral}
    Os capítulos 1--3 do livro de Vohra \cite{vohra2022dt} apresentam \textbf{três formas complementares} de classificar gêmeos digitais:

    \vspace{0.6em}
    \begin{columns}[T]
        \begin{column}{0.32\textwidth}
            \begin{block}{Taxonomia de Grieves}
                Classifica pela \textbf{relação} com o ativo físico:\\[0.3em]
                DTP, DTI, DTA e DTE
            \end{block}
        \end{column}
        \begin{column}{0.32\textwidth}
            \begin{block}{Por Granularidade}
                Classifica pelo \textbf{escopo} do que é modelado:\\[0.3em]
                Componente $\to$ Produto $\to$ Sistema $\to$ Processo
            \end{block}
        \end{column}
        \begin{column}{0.32\textwidth}
            \begin{block}{Nível de Integração}
                Classifica pelo \textbf{fluxo de dados} entre físico e virtual:\\[0.3em]
                Modelo $\to$ Sombra $\to$ Gêmeo
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \footnotesize
    Fonte: Grieves \& Vickers \cite{grieves2017dt}; Vohra \cite{vohra2022dt}, caps.\ 1--3; Fuller et al.\ \cite{fuller2020dt}.
\end{frame}

% --- Slide 3.2: Taxonomia de Grieves (DTP / DTI) ---
\begin{frame}{Taxonomia de Grieves -- DTP e DTI}
    \footnotesize
    Proposta por Michael Grieves (2002) e formalizada com John Vickers (NASA, 2010) \cite{grieves2017dt}:

    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{DTP -- Digital Twin Prototype}
                \footnotesize
                \begin{itemize}\setlength\itemsep{1pt}
                    \item Existe \textbf{antes} do ativo físico
                    \item Informações para \textbf{descrever e produzir} o ativo real
                    \item Modelo 3D anotado, especificações de materiais, \emph{bill of processes}, \emph{bill of disposal}
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{DTI -- Digital Twin Instance}
                \footnotesize
                \begin{itemize}\setlength\itemsep{1pt}
                    \item Vinculado a um ativo físico \textbf{específico} por toda sua vida útil
                    \item Modelo 3D com GD\&T, histórico de fabricação, testes, registros de manutenção
                    \item Dados de sensores: passados, presentes e preditos
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}

    \vspace{0.3em}
    \centering
    \scriptsize \textit{``DTPs devem existir para todos os produtos sofisticados; DTIs existem apenas quando é importante ter informação ao longo de toda a vida do produto.''} -- Grieves \cite{grieves2017dt}
\end{frame}

% --- Slide 3.3: Taxonomia de Grieves (DTA / DTE) ---
\begin{frame}{Taxonomia de Grieves -- DTA e DTE}
    \begin{columns}[T]
        \begin{column}{0.45\textwidth}
            \begin{block}{DTA -- Digital Twin Aggregate}
                \footnotesize
                \begin{itemize}\setlength\itemsep{1pt}
                    \item Agregação de \textbf{todos} os DTIs
                    \item Análise macro e correlação entre múltiplas instâncias
                    \item Ex: frota de aeronaves, parque eólico
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.50\textwidth}
            \begin{block}{DTE -- Digital Twin Environment}
                \footnotesize
                Espaço integrado multi-domínio para operar os gêmeos digitais:
                \begin{description}\setlength\itemsep{1pt}
                    \item[\textcolor{dexpiBlue}{Preditivo}] Prever comportamento e desempenho futuro
                    \item[\textcolor{dexpiBlue}{Interrogativo}] Consultar históricos passados e presentes
                \end{description}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.3em}
    \centering

\end{frame}

% --- Slide 3.6: Tabela Comparativa dos Níveis ---
\begin{frame}{Modelo $\times$ Sombra $\times$ Gêmeo -- Comparação}
    \vspace{0.3em}
    \centering
    \footnotesize
    \begin{tabular}{l c c c}
        \toprule
        \textbf{Característica} & \textbf{Modelo Digital} & \textbf{Sombra Digital} & \textbf{Gêmeo Digital} \\
        \midrule
        Fluxo de dados      & Nenhum           & Unidirecional          & \textbf{Bidirecional} \\
        Atualização automática & Não             & Sim (físico$\to$digital) & Sim (ambas direções)  \\
        Reflete estado real  & Não              & Sim                    & Sim \\
        Atua sobre o físico  & Não              & Não                    & \textbf{Sim} \\
        Exemplo              & Modelo CAD 3D    & Dashboard IoT          & Controle autônomo \\
        \bottomrule
    \end{tabular}

    \vspace{0.8em}
    \begin{alertblock}{Atenção}
        Muitos sistemas apresentados como ``gêmeos digitais'' na indústria são, na verdade, \textbf{sombras digitais} -- recebem dados do ativo físico, mas \textbf{não enviam comandos} de volta \cite{fuller2020dt}.
    \end{alertblock}
\end{frame}

% --- Slide 3.7: Ferramentas do Conhecimento (Cap. 2) ---
\begin{frame}{Gêmeo Digital como Ferramenta de Conhecimento}
    Grieves \cite{grieves2014whitepaper} identifica que o gêmeo digital suporta três ferramentas fundamentais do conhecimento humano:

    \vspace{0.5em}
    \begin{columns}[T]
        \begin{column}{0.31\textwidth}
            \begin{block}{Conceituação}
                Visualizar o comportamento do ativo de forma \textbf{intuitiva e visual}, ao invés de ler relatórios tabulares.
            \end{block}
        \end{column}
        \begin{column}{0.31\textwidth}
            \begin{block}{Comparação}
                Comparar o estado \textbf{ideal} com o \textbf{real}, utilizando \emph{tolerance corridors} para identificar desvios.
            \end{block}
        \end{column}
        \begin{column}{0.31\textwidth}
            \begin{block}{Colaboração}
                Compartilhar a mesma visualização com \textbf{múltiplos especialistas} em locais remotos, simultaneamente.
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \footnotesize
    Fonte: Grieves \cite{grieves2014whitepaper}; apresentado no Cap.\ 2 de Vohra \cite{vohra2022dt}.
\end{frame}

% --- Slide 3.8: Resumo da Seção ---
\begin{frame}{Tipos de Gêmeos Digitais -- Resumo}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Taxonomia de Grieves}
                \footnotesize
                \begin{description}\setlength\itemsep{0pt}
                    \item[DTP] Protótipo (pré-fabricação)
                    \item[DTI] Instância (ativo específico)
                    \item[DTA] Agregado (todos os DTIs)
                    \item[DTE] Ambiente de operação
                \end{description}
            \end{block}

            \vspace{0.2em}
            \begin{exampleblock}{Nível de Integração}
                \footnotesize
                \begin{description}\setlength\itemsep{0pt}
                    \item[Modelo] Sem fluxo de dados
                    \item[Sombra] Fluxo unidirecional
                    \item[Gêmeo] Fluxo bidirecional
                \end{description}
            \end{exampleblock}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Por Granularidade}
                \footnotesize
                \begin{description}\setlength\itemsep{0pt}
                    \item[Parts] Componente individual
                    \item[Product] Conjunto de componentes
                    \item[System] Conjunto de produtos
                    \item[Process] Manufatura completa
                \end{description}
            \end{block}

            \vspace{0.2em}
            \begin{alertblock}{Ponto-chave}
                \footnotesize
                As três classificações são \textbf{ortogonais}: um mesmo gêmeo digital pode ser um DTI, do tipo Process Twin, com integração bidirecional.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\section{Aplicações em diferentes setores}
%------------------------------------------------

\begin{frame}{4. Aplicações em Diferentes Setores}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Óleo e gás
                \item Energia
                \item Saúde

            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Cidades inteligentes
                \item Transporte
                \item Construção civil
            \end{itemize}
        \end{column}
    \end{columns}
    \vfill
\end{frame}

\begin{frame}{Indústria}
    \begin{itemize}
        \item A indústria utiliza gêmeos digitais para \textbf{analisar, monitorar e gerenciar} os sistemas produtivos de forma virtual
        \item Apoia a \textbf{previsão} de comportamentos futuros e de anomalias antes que ocorram
        \item Pilar da Indústria 4.0: integra TI, engenharia mecânica, eletrônica, mecatrônica e automação em um só modelo
        \item Depende de dados homogêneos e de um fluxo constante de dados entre o objeto físico e sua réplica digital
    \end{itemize}
    \begin{exampleblock}{Benefício central}
        Decisões industriais passam a ser guiadas por dados em tempo real, reduzindo custos operacionais e o tempo de resposta a falhas.
    \end{exampleblock}
\end{frame}

\begin{frame}{Óleo e Gás}
    \begin{itemize}
        \item Setor com ambientes de trabalho hostis e perigosos: o gêmeo digital ajuda a antecipar falhas e reduzir riscos de eventos indesejados
        \item Aplicações ao longo da cadeia:
        \begin{itemize}
            \item Planejamento e otimização do processo de perfuração
            \item Monitoramento de desempenho de campos petrolíferos
            \item Simulação e análise de dados de produção
            \item Segurança dos operadores 
            \item Manutenção preditiva de equipamentos
        \end{itemize}
        \item Benefícios: eficiência de produção, conformidade regulatória, redução de custos e maior segurança operacional
    \end{itemize}
\end{frame}

\begin{frame}{Energia}
    \begin{itemize}
        \item Otimização da manutenção e ciclo de vida de equipamentos como turbinas e geradores.
        \item Essa tecnologia também pode ser implementada em malhas de energia em cidades inteligentes para \textbf{otimizar a distribuição de energia}.
    \end{itemize}
    \begin{exampleblock}{Exemplos reais}
        \begin{itemize}
            \item \textbf{GE Digital Wind Farm}: gêmeo digital de parques eólicos que ajusta o projeto de cada turbina às condições específicas do local de instalação
            \item \textbf{Hitachi Energy -- IdentiQ}: gêmeo digital para sistemas de corrente contínua de alta tensão e qualidade de energia
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{Saúde -- Gestão Hospitalar e Aplicações Clínicas}
    \begin{itemize}
        \item \textbf{Gestão de fluxo hospitalar}: réplicas virtuais de unidades hospitalares permite testar mudanças de estratégia operacional, escala de equipe e modelos de atendimento antes de aplicá-los na prática
        \item \textbf{Pré-planejamento cirúrgico}: simulação prévia do procedimento em modelo do paciente, treinamento de equipes e redução do risco cirúrgico
    \end{itemize}
    \begin{exampleblock}{Exemplos reais}
        \begin{itemize}
            \item \textbf{GE -- Hospital of Future (HoF)}: suíte de simulação para prever a operação hospitalar e o cuidado ao paciente
            \item \textbf{Mater Hospital (Dublin)} com Siemens Healthineers: gêmeo digital do departamento de radiologia que resultou em menor tempo de espera, maior utilização de equipamentos e menor custo de pessoal
        \end{itemize}
    \end{exampleblock}
\end{frame}
\begin{frame}{Cidades Inteligentes}
    \begin{itemize}
        \item \textbf{Gestão de tráfego}: simulação do deslocamento de pessoas e veículos para identificar gargalos e testar soluções de fluxo
        \item \textbf{Construção}: gêmeos digitais orientam projeto, obra e operação de edificações urbanas
        \item \textbf{Monitoramento estrutural}: pontes e prédios acompanhados em tempo real para prevenir falhas
        \item \textbf{Escoamento}: previsão e monitoramento de enchentes em tempo real
    \end{itemize}
\end{frame}

\begin{frame}{Transporte}
    \begin{exampleblock}{Aviação}
        \begin{itemize}
            \item \textbf{Rolls-Royce}: gêmeos digitais de motores aeronáuticos para manutenção preventiva
            \item \textbf{GE Aviation} (Azure Digital Twin): acompanhamento em tempo real do estado da aeronave e de seus componentes, antecipando reparos
        \end{itemize}
    \end{exampleblock}
    \begin{exampleblock}{Setor automotivo}
        \begin{itemize}
            \item \textbf{Bridgestone} -- \textit{Virtual Tire Modelling}: reduz o número de protótipos físicos para produção de pneus
            \item \textbf{Tesla}: cada veículo vendido possui um gêmeo digital, usado para orientar atualizações de software via dados dos sensores
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{Construção Civil}
    \begin{itemize}
        \item Atua nas três fases do ciclo de vida da obra:
        \begin{itemize}
            \item \textbf{Projeto}: testa o impacto de mudanças de design antes da execução
            \item \textbf{Construção}: monitoramento remoto do canteiro, antecipando erros e riscos
            \item \textbf{Operação e manutenção}: acompanha a saúde estrutural do edifício já construído
        \end{itemize}
        \item \textbf{Segurança do trabalhador}: identifica riscos em tempo real no canteiro de obras
    \end{itemize}
    \begin{exampleblock}{Exemplos reais}
        \begin{itemize}
        \item \textbf{Lendlease Group} testou a viabilidade de torres de 28--29 andares em madeira sustentável; 
        \item \textbf{CadMakers} projetou um edifício híbrido de madeira maciça de 18 andares -- ambos usando gêmeos digitais na fase de projeto.
        \end{itemize}
    \end{exampleblock}
\end{frame}

\textbf{\large Benefícios}

\vspace{0.2cm}

\begin{itemize}
    \item \textbf{Monitoramento}: Torna possível monitorar e manter a vigilância de um objeto, produto ou processo real de forma remota.
    \item \textbf{Simulação}: Permite analisar o comportamento do sistema e testar diferentes abordagens, obtendo insights valiosos aplicáveis ao mundo real.
    \item \textbf{Otimização}: Ao receber dados em tempo real, o modelo virtual se atualiza dinamicamente e permite controlar e otimizar a si mesmo.
\end{itemize}

\begin{frame}{5. Benefícios e Desafios de Implementação}

\textbf{\large Desafios}

\vspace{0.2cm}

\begin{itemize}
    \item \textbf{Integração de dados}: Um \textit{sensor defeituoso} ou a quebra de um link pode prejudicar severamente o funcionamento do sistema.
    \item \textbf{Interoperabilidade}: Sem o \textit{fluxo contínuo de dados} estabelecido entre o mundo físico e o virtual, o gêmeo digital não consegue atualizar corretamente.
    \item \textbf{Validação dos modelos}: O modelo matemático e comportamental no ambiente virtual precisa \textit{coincidir exatamente com o comportamento do objeto físico}.
\end{itemize}
